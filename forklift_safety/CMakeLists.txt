cmake_minimum_required(VERSION 3.8)
project(forklift_safety)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(forklift_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(yaml-cpp REQUIRED)

set(dependencies rclcpp forklift_msgs geometry_msgs nav_msgs nav2_msgs sensor_msgs std_msgs tf2_ros)
add_executable(safety_command_gate src/safety_command_gate.cpp)
target_compile_features(safety_command_gate PRIVATE cxx_std_17)
target_compile_options(safety_command_gate PRIVATE -Wall -Wextra -Wpedantic -O2)
target_include_directories(safety_command_gate PRIVATE include)
ament_target_dependencies(safety_command_gate ${dependencies})
target_link_libraries(safety_command_gate yaml-cpp)
install(TARGETS safety_command_gate DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)

# Keep the Python reference implementation available for offline comparison.
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/safety_command_gate_python DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_collision_core test/test_collision_core.cpp)
  target_include_directories(test_collision_core PRIVATE include)
  target_compile_features(test_collision_core PRIVATE cxx_std_17)
  ament_target_dependencies(test_collision_core ${dependencies})
  file(GLOB python_tests test/test_*.py)
  foreach(test_file ${python_tests})
    get_filename_component(test_name ${test_file} NAME_WE)
    if(NOT test_name MATCHES "^test_cpp_")
      ament_add_pytest_test(${test_name} ${test_file}
        APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}:${CMAKE_CURRENT_SOURCE_DIR}/test TIMEOUT 120)
    endif()
  endforeach()
  add_executable(collision_parity_runner test/collision_parity_runner.cpp)
  target_include_directories(collision_parity_runner PRIVATE include)
  target_compile_features(collision_parity_runner PRIVATE cxx_std_17)
  target_compile_options(collision_parity_runner PRIVATE -O2)
  ament_target_dependencies(collision_parity_runner ${dependencies})
  target_link_libraries(collision_parity_runner yaml-cpp)
  ament_add_pytest_test(cpp_parity test/test_cpp_parity.py
    APPEND_ENV PYTHONPATH=${CMAKE_CURRENT_SOURCE_DIR}
    ENV SAFETY_PARITY_RUNNER=$<TARGET_FILE:collision_parity_runner> TIMEOUT 120)
  ament_add_pytest_test(cpp_node test/test_cpp_node.py
    ENV SAFETY_CPP_NODE=$<TARGET_FILE:safety_command_gate> TIMEOUT 120)
endif()
ament_package()
